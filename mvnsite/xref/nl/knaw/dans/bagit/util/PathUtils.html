16  package nl.knaw.dans.bagit.util;
17  
18  import java.io.IOException;
19  import java.nio.file.Files;
20  import java.nio.file.Path;
21  import java.nio.file.attribute.DosFileAttributes;
22  
23  import nl.knaw.dans.bagit.domain.Bag;
24  import nl.knaw.dans.bagit.domain.Version;
25  import nl.knaw.dans.bagit.verify.FileCountAndTotalSizeVistor;
26  
27  /**
28   * Convenience class for dealing with various path issues
29   */
30  public final class PathUtils {
31    private static final String PAYLOAD_DIR_NAME = "data";
32    
33    //@Incubating
34   private static final String DOT_BAGIT_DIR_NAME = ".bagit";
35    
36    private PathUtils(){
37      //intentionally left blank
38    }
39  
40    /**
41     * Needed to get rid of findbugs "dodgy code warnings" in regards to getting the filename of a path as a string
42     * 
43     * @param path the path that you which to get the filename as a string
44     * @return the filename or an empty string
45     */
46    public static String getFilename(final Path path){
47      String filename = "";
48      if(path != null){
49        final Path filenamePath = path.getFileName();
50        if(filenamePath != null){
51          filename = filenamePath.toString();
52        }
53      }
54      
55      return filename;
56    }
57    
58    /**
59     * as per https://github.com/jkunze/bagitspec/commit/152d42f6298b31a4916ea3f8f644ca4490494070 decode percent encoded filenames
60     * @param encoded the encoded filename
61     * @return the decoded filename 
62     */
63    public static String decodeFilname(final String encoded){
64      return encoded.replaceAll("%0A", "\n").replaceAll("%0D", "\r");
65    }
66    
67    /**
68     * as per https://github.com/jkunze/bagitspec/commit/152d42f6298b31a4916ea3f8f644ca4490494070 encode any new lines or carriage returns
69     * @param path the path to encode
70     * @return the encoded filename
71     */
72    public static String encodeFilename(final Path path){
73      return path.toString().replaceAll("\n", "%0A").replaceAll("\r", "%0D");
74    }
75    
76    /**
77     * Due to the way that windows handles hidden files vs. *nix 
78     * we use this method to determine if a file or folder is really hidden
79     * @param path the file or folder to check if hidden
80     * @return if the file or folder is hidden
81     * @throws IOException if there is an error reading the file/folder
82     */
83    public static boolean isHidden(final Path path) throws IOException{
84      //cause Files.isHidden() doesn't work properly for windows if the file is a directory
85      if (System.getProperty("os.name").contains("Windows")){
86        return Files.readAttributes(path, DosFileAttributes.class).isHidden();
87      }
88  
89      return Files.isHidden(path);
90    }
91    
92    /**
93     * With bagit version 2.0 (.bagit)
94     * payload files are no longer in the "data" directory. This method accounts for this
95     * and will return the directory that contains the payload files
96     * 
97     * @param bag that contains the payload files you want
98     * @return the directory that contains the payload files 
99     */
100   public static Path getDataDir(final Bag bag){
101     if(bag.getVersion().isSameOrNewer(new Version(2, 0))){ //is it a .bagit version?
102       return bag.getRootDir();
103     }
104     
105     return bag.getRootDir().resolve(PAYLOAD_DIR_NAME);
106   }
107   
108   /**
109    * With bagit version 2.0 (.bagit)
110    * payload files are no longer in the "data" directory. This method accounts for this
111    * and will return the directory that contains the payload files
112    * 
113    * @param version the bag version
114    * @param output where the bag is being or was written to
115    * 
116    * @return the payload directory for the output directory
117    */
118   public static Path getDataDir(final Version version, final Path output){
119     if(version.isSameOrNewer(new Version(2, 0))){ //is it a .bagit version?
120       return output;
121     }
122     
123     return output.resolve(PAYLOAD_DIR_NAME);
124   }
125   
126   /**
127    * With bagit version 2.0 (.bagit) bagit specific files are no longer at the bag root directory.
128    * This method accounts for this and will return the directory that contains the bag specific files.
129    * 
130    * @param bag the bag
131    * 
132    * @return the directory which contains the bag specific files, like manifests or bagit.txt
133    */
134   public static Path getBagitDir(final Bag bag){
135     if(bag.getVersion().isSameOrNewer(new Version(2, 0))){ //is it a .bagit version?
136       return bag.getRootDir().resolve(DOT_BAGIT_DIR_NAME);
137     }
138     
139     return bag.getRootDir();
140   }
141   
142   /**
143    * With bagit version 2.0 (.bagit) bagit specific files are no longer at the bag root directory.
144    * This method accounts for this and will return the directory that contains the bag specific files.
145    * 
146    * @param version the bag version
147    * @param bagRoot the root directory of the bag
148    * 
149    * @return the directory which contains the bag specific files, like manifests or bagit.txt
150    */
151   public static Path getBagitDir(final Version version, final Path bagRoot){
152     if(version.isSameOrNewer(new Version(2, 0))){ //is it a .bagit version?
153       return bagRoot.resolve(DOT_BAGIT_DIR_NAME);
154     }
155     
156     return bagRoot;
157   }
158   
159   /**
160    * Calculate the total file and byte count of the files in the payload directory
161    * 
162    * @param dataDir the directory to calculate the payload-oxum
163    * 
164    * @return the string representation of the payload-oxum value
165    * 
166    * @throws IOException if there is an error reading any of the files
167    */
168   public static String generatePayloadOxum(final Path dataDir) throws IOException{
169     final FileCountAndTotalSizeVistor visitor = new FileCountAndTotalSizeVistor();
170     
171     Files.walkFileTree(dataDir, visitor);
172     
173     return visitor.getTotalSize() + "." + visitor.getCount();
174   }
175 }
