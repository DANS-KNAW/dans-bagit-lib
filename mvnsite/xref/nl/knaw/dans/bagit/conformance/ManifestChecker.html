45  /**
46   * Part of the BagIt conformance suite. 
47   * This checker checks for various problems related to the manifests in a bag.
48   */
49  //TODO refactor to remove PMD warnings!
50  @SuppressWarnings({"PMD.UseLocaleWithCaseConversions", "PMD.TooManyMethods", "PMD.GodClass"})
51  public final class ManifestChecker {
52    private static final Logger logger = LoggerFactory.getLogger(ManifestChecker.class);
53    private static final ResourceBundle messages = ResourceBundle.getBundle("MessageBundle");
54    
55    private static final String THUMBS_DB_FILE = "[Tt][Hh][Uu][Mm][Bb][Ss]\\.[Dd][Bb]";
56    private static final String DS_STORE_FILE = "\\.[Dd][Ss]_[Ss][Tt][Oo][Rr][Ee]";
57    private static final String SPOTLIGHT_FILE = "\\.[Ss][Pp][Oo][Tt][Ll][Ii][Gg][Hh][Tt]-[Vv]100";
58    private static final String TRASHES_FILE = "\\.(_.)?[Tt][Rr][Aa][Ss][Hh][Ee][Ss]";
59    private static final String FS_EVENTS_FILE = "\\.[Ff][Ss][Ee][Vv][Ee][Nn][Tt][Ss][Dd]";
60    private static final String OS_FILES_REGEX = ".*data/(" + THUMBS_DB_FILE + "|" + DS_STORE_FILE + "|" + SPOTLIGHT_FILE + "|" + TRASHES_FILE + "|" + FS_EVENTS_FILE + ")";
61    private static final Version VERSION_1_0 = new Version(1,0);
62    
63    private ManifestChecker(){
64      //intentionally left empty
65    }

67    /**
68     * Check for all the manifest specific potential problems
69     * 
70     * @param version the version of the bag we are checking
71     * @param bagitDir the directory where the manifests are stored
72     * @param encoding the encoding of the manifests
73     * @param warnings the set of warnings that will be appended to while checking
74     * @param warningsToIgnore the set of warnings to ignore
75     * 
76     * @throws IOException if there is a problem reading a file (because it doesn't exist) 
77     * @throws InvalidBagitFileFormatException if one (or more) of the files does not match the formatting as specified in the specification
78     * @throws MaliciousPathException if someone crafted the bag to specifically try and write outside the bag directory
79     * @throws UnsupportedAlgorithmException if a manifest uses an algorithm that the computer doesn't know how to use
80     */
81    public static void checkManifests(final Version version, final Path bagitDir, final Charset encoding, final Set<BagitWarning> warnings, 
82        final Collection<BagitWarning> warningsToIgnore) throws IOException, InvalidBagitFileFormatException, MaliciousPathException, UnsupportedAlgorithmException{
83          
84      boolean missingTagManifest = true;
85      final List<Path> payloadManifests = new ArrayList<>();
86      final List<Path> tagManifests = new ArrayList<>();
87      try(final DirectoryStream<Path> files = Files.newDirectoryStream(bagitDir)){
88        for(final Path file : files){
89          missingTagManifest = missingTagManifest && checkManifest(file, payloadManifests, tagManifests, encoding, warnings, warningsToIgnore);
90        }
91      }
92      
93      if(!warnings.contains(BagitWarning.MANIFEST_SETS_DIFFER)){
94        checkManifestSets(version, tagManifests, payloadManifests, warnings, encoding);
95      }
96      
97      if(!warningsToIgnore.contains(BagitWarning.MISSING_TAG_MANIFEST) && missingTagManifest){
98        logger.warn(messages.getString("bag_missing_tag_manifest_warning"), bagitDir);
99        warnings.add(BagitWarning.MISSING_TAG_MANIFEST);
100     }
101   }
102   
103   private static boolean checkManifest(final Path file, final List<Path> payloadManifests, final List<Path> tagManifests, 
104       final Charset encoding, final Set<BagitWarning> warnings, 
105       final Collection<BagitWarning> warningsToIgnore) throws IOException, InvalidBagitFileFormatException{
106     boolean missingTagManifest = true;
107     final String filename = PathUtils.getFilename(file);
108     if(filename.contains("manifest-")){
109       if(filename.startsWith("manifest-")){
110         payloadManifests.add(file);
111         checkManifestPayload(file, encoding, warnings, warningsToIgnore, true);
112       }
113       else{
114         tagManifests.add(file);
115         checkManifestPayload(file, encoding, warnings, warningsToIgnore, false);
116         missingTagManifest = false;
117       }
118       
119       final String algorithm = filename.split("[-\\.]")[1];
120       checkAlgorthm(algorithm, warnings, warningsToIgnore);
121     }
122     
123     return missingTagManifest;
124   }

126   /*
127    * Check for a "bag within a bag", relative paths, and OS specific files in the manifests
128    */
129   private static void checkManifestPayload(final Path manifestFile, final Charset encoding, final Set<BagitWarning> warnings, 
130       final Collection<BagitWarning> warningsToIgnore, final boolean isPayloadManifest) 
131           throws IOException, InvalidBagitFileFormatException{
132     
133     try(final BufferedReader reader = Files.newBufferedReader(manifestFile, encoding)){
134       final Set<String> paths = new HashSet<>();
135       
136       String line = reader.readLine();
137       while(line != null){
138         String path = parsePath(line);
139         
140         path = checkForManifestCreatedWithMD5SumTools(path, warnings, warningsToIgnore);
141         
142         checkForDifferentCase(path, paths, manifestFile, warnings, warningsToIgnore);
143         paths.add(path.toLowerCase());
144         
145         if(encoding.name().startsWith("UTF")){
146           checkNormalization(path, manifestFile.getParent(), warnings, warningsToIgnore);
147         }
148         checkForBagWithinBag(line, warnings, warningsToIgnore, isPayloadManifest);
149         checkForRelativePaths(line, warnings, warningsToIgnore, manifestFile);
150         checkForOSSpecificFiles(line, warnings, warningsToIgnore, manifestFile);
151         
152         line = reader.readLine();
153       }
154     }
155   }
156   
157   /*
158    * Check to make sure it conforms to <hash> <path>
159    */
160   static String parsePath(final String line) throws InvalidBagitFileFormatException{
161     final String[] parts = line.split("\\s+", 2);
162     if(parts.length < 2){
163       final String formattedMessage = messages.getString("manifest_line_violated_spec_error");
164       throw new InvalidBagitFileFormatException(MessageFormatter.format(formattedMessage, line).getMessage());
165     }
166     
167     return parts[1];
168   }

170   /*
171    * We allow for MD5sum tools for compatibility but it is not recommended
172    */
173   private static String checkForManifestCreatedWithMD5SumTools(final String path, final Set<BagitWarning> warnings, final Collection<BagitWarning> warningsToIgnore){
174     String fixedPath = path;
175     final boolean startsWithStar = path.charAt(0) == '*';
176     
177     if(startsWithStar){
178       fixedPath = path.substring(1);
179     }
180     
181     if(!warningsToIgnore.contains(BagitWarning.MD5SUM_TOOL_GENERATED_MANIFEST) && startsWithStar){
182       logger.warn(messages.getString("md5sum_generated_line_warning"), path);
183       warnings.add(BagitWarning.MD5SUM_TOOL_GENERATED_MANIFEST);
184     }
185     
186     return fixedPath;
187   }

189   /*
190    * Check that the same line doesn't already exist in the set of paths
191    */
192   private static void checkForDifferentCase(final String path, final Set<String> paths, final Path manifestFile, 
193       final Set<BagitWarning> warnings, final Collection<BagitWarning> warningsToIgnore){
194     if(!warningsToIgnore.contains(BagitWarning.DIFFERENT_CASE) && paths.contains(path.toLowerCase())){
195       logger.warn(messages.getString("different_case_warning"), manifestFile, path);
196       warnings.add(BagitWarning.DIFFERENT_CASE);
197     }
198   }

200   /*
201    * Check that the file specified has not changed its normalization (i.e. have the bytes changed but it still looks the same?)
202    */
203   private static void checkNormalization(final String path, final Path rootDir, final Set<BagitWarning> warnings, final Collection<BagitWarning> warningsToIgnore) throws IOException{
204     if(!warningsToIgnore.contains(BagitWarning.DIFFERENT_NORMALIZATION)){
205       
206       final Path fileToCheck = rootDir.resolve(path).normalize();
207       final Path dirToCheck = fileToCheck.getParent();
208       if(dirToCheck == null){ 
209         final String formattedMessage = messages.getString("cannot_access_parent_path_error");
210         throw new IOException(MessageFormatter.format(formattedMessage, fileToCheck).getMessage()); //to satisfy findbugs
211       }
212       final String normalizedFileToCheck = normalizePathToNFD(fileToCheck);
213       
214       try(final DirectoryStream<Path> files = Files.newDirectoryStream(dirToCheck)){
215         for(final Path file : files){
216           final String normalizedFile = normalizePathToNFD(file);
217           
218           if(!file.equals(fileToCheck) && normalizedFileToCheck.equals(normalizedFile)){
219             logger.warn(messages.getString("different_normalization_in_manifest_warning"), fileToCheck);
220             warnings.add(BagitWarning.DIFFERENT_NORMALIZATION);
221           }
222         }
223       }
224     }
225   }
226   
227   /*
228    * Normalize to Canonical decomposition.
229    */
230   static String normalizePathToNFD(final Path path){
231     return Normalizer.normalize(path.toString(), Normalizer.Form.NFD);
232   }
233   
234   /*
235    * check for a bag within a bag
236    */
237   private static void checkForBagWithinBag(final String line, final Set<BagitWarning> warnings, final Collection<BagitWarning> warningsToIgnore, final boolean isPayloadManifest){
238     if(!warningsToIgnore.contains(BagitWarning.BAG_WITHIN_A_BAG) && isPayloadManifest && line.contains("manifest-")){
239       logger.warn(messages.getString("bag_within_bag_warning"));
240       warnings.add(BagitWarning.BAG_WITHIN_A_BAG);
241     }
242   }
243   
244   /*
245    * Check for relative paths (i.e. ./) in the manifest
246    */
247   private static void checkForRelativePaths(final String line, final Set<BagitWarning> warnings, final Collection<BagitWarning> warningsToIgnore, final Path manifestFile){
248     if(!warningsToIgnore.contains(BagitWarning.LEADING_DOT_SLASH) && line.contains("./")){
249       logger.warn(messages.getString("leading_dot_slash_warning"), manifestFile, line);
250       warnings.add(BagitWarning.LEADING_DOT_SLASH);
251     }
252   }
253   
254   /*
255    * like .DS_Store or Thumbs.db
256    */
257   private static void checkForOSSpecificFiles(final String line, final Set<BagitWarning> warnings, final Collection<BagitWarning> warningsToIgnore, final Path manifestFile){
258     if(!warningsToIgnore.contains(BagitWarning.OS_SPECIFIC_FILES) && line.matches(OS_FILES_REGEX)){
259       logger.warn(messages.getString("os_specific_files_warning"), manifestFile, line);
260       warnings.add(BagitWarning.OS_SPECIFIC_FILES);
261     }
262   }
263   
264   /*
265    * Check for anything weaker than SHA-512
266    */
267   static void checkAlgorthm(final String algorithm, final Set<BagitWarning> warnings, final Collection<BagitWarning> warningsToIgnore){
268     final String upperCaseAlg = algorithm.toUpperCase();
269     if(!warningsToIgnore.contains(BagitWarning.WEAK_CHECKSUM_ALGORITHM) && 
270         (upperCaseAlg.startsWith("MD") || upperCaseAlg.matches("SHA(1|224|256|384)?"))){
271       logger.warn(messages.getString("weak_algorithm_warning"), algorithm);
272       warnings.add(BagitWarning.WEAK_CHECKSUM_ALGORITHM);
273     }
274     
275     else if(!warningsToIgnore.contains(BagitWarning.NON_STANDARD_ALGORITHM) && !"SHA512".equals(upperCaseAlg)){
276       logger.warn(messages.getString("non_standard_algorithm_warning"), algorithm);
277       warnings.add(BagitWarning.NON_STANDARD_ALGORITHM);
278     }
279   }

281   static void checkManifestSets(final Version version, final List<Path> tagManifests, final List<Path> payloadManifests, 
282       final Set<BagitWarning> warnings, final Charset encoding) 
283           throws IOException, MaliciousPathException, UnsupportedAlgorithmException, InvalidBagitFileFormatException{
284   //edge case, for version 1.0+ all tag manifests SHOULD list the same set of files
285     if(tagManifests.size() > 1 && VERSION_1_0.isSameOrOlder(version)){
286       checkManifestsListSameSetOfFiles(warnings, tagManifests, encoding);
287     }
288     
289     //edge case, for version 1.0+ all payload manifests SHOULD list the same set of files
290     if(payloadManifests.size() > 1 && VERSION_1_0.isSameOrOlder(version)){
291       checkManifestsListSameSetOfFiles(warnings, payloadManifests, encoding);
292     }
293   }
294   
295   //starting with version 1.0 all manifest types (tag, payload) should list the same set of files
296   @SuppressWarnings("PMD.EmptyCatchBlock")
297   static void checkManifestsListSameSetOfFiles(final Set<BagitWarning> warnings, final List<Path> manifestPaths, final Charset charset) throws IOException, MaliciousPathException, UnsupportedAlgorithmException, InvalidBagitFileFormatException{
298     final StandardBagitAlgorithmNameToSupportedAlgorithmMapping nameMapping = new StandardBagitAlgorithmNameToSupportedAlgorithmMapping();
299     
300     Manifest compareToManifest = null;
301     Path compareToManifestPath = null;
302     for (final Path manifestPath : manifestPaths) {
303       try {
304         final Manifest manifest = ManifestReader.readManifest(nameMapping, manifestPath, manifestPath.getParent(), charset);
305         if(compareToManifest == null) {
306           compareToManifestPath = manifestPath;
307           compareToManifest = manifest;
308           continue;
309         }
310         
311         if(!compareToManifest.getFileToChecksumMap().keySet().equals(manifest.getFileToChecksumMap().keySet())) {
312           logger.warn(messages.getString("manifest_fileset_differ"), compareToManifestPath, manifestPath);
313           warnings.add(BagitWarning.MANIFEST_SETS_DIFFER);
314         }
315       }
316       catch(UnsupportedAlgorithmException e) {
317         //ignore an unsupported algorithm as it is caught in checkAlgorthm()
318       }
319     }
320   }
