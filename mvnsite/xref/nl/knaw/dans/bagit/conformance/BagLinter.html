57  /**
58   * Responsible for checking a bag and providing insight into how it cause problems.
59   * This class is only to be used on VALID bags, using it on un-validated bags may result in
60   * exceptions being thrown (like {@link java.io.IOException} )
61   */
62  public final class BagLinter {
63    private static final Logger logger = LoggerFactory.getLogger(BagLinter.class);
64    private static final ResourceBundle messages = ResourceBundle.getBundle("MessageBundle");
65    private static final Version VERSION_1_0 = new Version(1,0);
66    
67    private BagLinter(){
68  //    intentionally left empty
69    }

71    /**
72     * Check a bag against a bagit-profile as described by 
73     * <a href="https://github.com/ruebot/bagit-profiles">https://github.com/ruebot/bagit-profiles</a>
74     * <br>Note: <b> This implementation does not check the Serialization part of the profile!</b>
75     * 
76     * @param jsonProfile the input stream to the json string describing the profile
77     * @param bag the bag to check against the profile
78     * 
79     * @throws IOException if there is a problem reading the profile or some of the bag files
80     * @throws JsonMappingException if there is a problem mapping the profile to the {@link BagitProfile}
81     * @throws JsonParseException if there is a problem parsing the json while mapping to java object
82     * 
83     * @throws FetchFileNotAllowedException if there is a fetch file when the profile prohibits it
84     * @throws MetatdataValueIsNotAcceptableException if a metadata value is not in the list of acceptable values
85     * @throws MetatdataValueIsNotRepeatableException if a metadata value shows up more than once when not repeatable
86     * @throws RequiredMetadataFieldNotPresentException if a metadata field is not present but it should be
87     * @throws RequiredManifestNotPresentException if a payload or tag manifest type is not present but should be
88     * @throws BagitVersionIsNotAcceptableException if the version of the bag is not in the list of acceptable versions
89     * @throws RequiredTagFileNotPresentException if a tag file is not present but should be
90     */
91    public static void checkAgainstProfile(final InputStream jsonProfile, final Bag bag) throws JsonParseException, JsonMappingException,
92    IOException, FetchFileNotAllowedException, RequiredMetadataFieldNotPresentException, MetatdataValueIsNotAcceptableException, RequiredManifestNotPresentException, 
93    BagitVersionIsNotAcceptableException, RequiredTagFileNotPresentException, MetatdataValueIsNotRepeatableException{
94      BagProfileChecker.bagConformsToProfile(jsonProfile, bag);
95    }

97    /**
98     * The BagIt specification is very flexible in what it allows which leads to situations 
99     * where something may be technically allowed, but should be discouraged.
100    * This method checks a bag for potential problems, or other items that are allowed but discouraged.
101    * This <strong>does not</strong> validate a bag. See {@link BagVerifier} instead.
102    * 
103    * @param rootDir the root directory of the bag
104    * 
105    * @return a set of {@link BagitWarning} detailing all items that should be fixed.
106    * 
107    * @throws InvalidBagMetadataException if the bag metadata does not conform to the bagit specification
108    * @throws UnparsableVersionException if there is an error reading the bagit version
109    * @throws IOException if there was an error reading a file
110    * @throws UnsupportedAlgorithmException if there is an error while reading one of the manifests due to the algorithm being unsupported
111    * @throws MaliciousPathException if the path is crafted to be malicious (overwrite non bag files)
112    */
113   public static Set<BagitWarning> lintBag(final Path rootDir) throws IOException, UnparsableVersionException, InvalidBagMetadataException, InvalidBagitFileFormatException, MaliciousPathException, UnsupportedAlgorithmException{
114     return lintBag(rootDir, Collections.emptyList());
115   }

117   /**
118    * The BagIt specification is very flexible in what it allows which leads to situations 
119    * where something may be technically allowed, but should be discouraged.
120    * This method checks a bag for potential problems, or other items that are allowed but discouraged.
121    * This <strong>does not</strong> validate a bag. See {@link BagVerifier} instead.
122    * 
123    * @param rootDir the root directory of the bag
124    * @param warningsToIgnore any {@link BagitWarning} to ignore when linting
125    * 
126    * @return a set of {@link BagitWarning} detailing all items that should be fixed.
127    * 
128    * @throws InvalidBagMetadataException if the bag metadata does not conform to the bagit specification
129    * @throws UnparsableVersionException if there is an error reading the bagit version
130    * @throws IOException if there was an error reading a file
131    * @throws UnsupportedAlgorithmException if there is an error while reading one of the manifests due to the algorithm being unsupported
132    * @throws MaliciousPathException if the path is crafted to be malicious (overwrite non bag files)
133    */
134   public static Set<BagitWarning> lintBag(final Path rootDir, final Collection<BagitWarning> warningsToIgnore) throws IOException, UnparsableVersionException, InvalidBagMetadataException, InvalidBagitFileFormatException, MaliciousPathException, UnsupportedAlgorithmException{
135     final Set<BagitWarning> warnings = new HashSet<>();
136     
137     //@Incubating
138     Path bagitDir = rootDir.resolve(".bagit");
139     if(!Files.exists(bagitDir)){
140       bagitDir = rootDir;
141     }
142     
143     final Path bagitFile = bagitDir.resolve("bagit.txt");
144     checkForExtraLines(bagitFile, warnings, warningsToIgnore);
145     final SimpleImmutableEntry<Version, Charset> bagitInfo = BagitTextFileReader.readBagitTextFile(bagitFile);
146     
147     logger.info(messages.getString("checking_encoding_problems"));
148     EncodingChecker.checkEncoding(bagitInfo.getValue(), warnings, warningsToIgnore);
149     
150     logger.info(messages.getString("checking_latest_version"));
151     VersionChecker.checkVersion(bagitInfo.getKey(), warnings, warningsToIgnore);
152     
153     logger.info(messages.getString("checking_manifest_problems"));
154     ManifestChecker.checkManifests(bagitInfo.getKey(), bagitDir, bagitInfo.getValue(), warnings, warningsToIgnore);
155 
156     logger.info(messages.getString("checking_metadata_problems"));
157     MetadataChecker.checkBagMetadata(bagitDir, bagitInfo.getValue(), warnings, warningsToIgnore);
158     
159     return warnings;
160   }
161   
162   private static void checkForExtraLines(final Path bagitFile, final Collection<BagitWarning> warnings, final Collection<BagitWarning> warningsToIgnore) throws InvalidBagMetadataException, IOException, UnparsableVersionException{
163     if(warningsToIgnore.contains(BagitWarning.EXTRA_LINES_IN_BAGIT_FILES)){
164       logger.debug(messages.getString("skipping_check_extra_lines"));
165       return;
166     }
167     
168     logger.debug(messages.getString("checking_extra_lines"));
169     final List<SimpleImmutableEntry<String, String>> pairs = KeyValueReader.readKeyValuesFromFile(bagitFile, ":", StandardCharsets.UTF_8);
170      
171     for(final SimpleImmutableEntry<String, String> pair : pairs){
172       if("BagIt-Version".equals(pair.getKey())){
173         final Version version = BagitTextFileReader.parseVersion(pair.getValue());
174         //versions before 1.0 specified it must be exactly 2 lines
175         if(pairs.size() > 2 && version.isOlder(VERSION_1_0)){
176           logger.warn(messages.getString("extra_lines_warning"), pairs.size());
177           warnings.add(BagitWarning.EXTRA_LINES_IN_BAGIT_FILES);
178         }
179       }
180     }
181   }
