16  package nl.knaw.dans.bagit.conformance;
17  
18  import java.util.ResourceBundle;
19  
20  /**
21   * The BagIt specification is very flexible in what it allows. 
22   * This leads to situations where something may be technically allowed, but should be discouraged. 
23   * This class is for that purpose, to allow reporting of these allowed but discouraged situations to the user.
24   */
25  public enum BagitWarning {
26    BAG_WITHIN_A_BAG("bag_within_a_bag"),
27    DIFFERENT_CASE("different_case"),
28    DIFFERENT_NORMALIZATION("different_normalization"),
29    EXTRA_LINES_IN_BAGIT_FILES("extra_lines_in_bagit_files"),
30    LEADING_DOT_SLASH("leading_dot_slash"),
31    NON_STANDARD_ALGORITHM("non_standard_algorithm"),
32    MD5SUM_TOOL_GENERATED_MANIFEST("md5sum_tool_generated_manifest"),
33    MISSING_TAG_MANIFEST("missing_tag_manifest"),
34    OLD_BAGIT_VERSION("old_bagit_version"),
35    OS_SPECIFIC_FILES("os_specific_files"),
36    PAYLOAD_OXUM_MISSING("payload_oxum_missing"),
37    TAG_FILES_ENCODING("tag_files_encoding"),
38    WEAK_CHECKSUM_ALGORITHM("weak_checksum_algorithm"),
39    MANIFEST_SETS_DIFFER("manifest_file_sets_differ_between_algorithms");
40    
41    private final String messageBundleKey;
42    private static final ResourceBundle messages = ResourceBundle.getBundle("MessageBundle");
43    
44    private BagitWarning(final String reason){
45      this.messageBundleKey = reason;
46    }
47  
48    public String getReason() {
49      return messages.getString(messageBundleKey);
50    }
51  }
