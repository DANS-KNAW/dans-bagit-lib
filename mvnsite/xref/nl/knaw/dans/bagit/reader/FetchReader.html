18  import java.io.BufferedReader;
19  import java.io.IOException;
20  import java.net.URL;
21  import java.nio.charset.Charset;
22  import java.nio.file.Files;
23  import java.nio.file.Path;
24  import java.util.ArrayList;
25  import java.util.List;
26  import java.util.ResourceBundle;
27  
28  import nl.knaw.dans.bagit.exceptions.InvalidBagitFileFormatException;
29  import org.slf4j.Logger;
30  import org.slf4j.LoggerFactory;
31  
32  import nl.knaw.dans.bagit.domain.FetchItem;
33  import nl.knaw.dans.bagit.exceptions.MaliciousPathException;
34  
35  /**
36   * This class is responsible for reading and parsing fetch.txt file from the filesystem
37   */
38  public final class FetchReader {
39    private static final Logger logger = LoggerFactory.getLogger(FetchReader.class);
40    private static final ResourceBundle messages = ResourceBundle.getBundle("MessageBundle");
41    private static final String FETCH_LINE_REGEX = ".*[ \t]*(\\d*|-)[ \t]*.*";
42  
43    private FetchReader(){
44      //intentionally left empty
45    }
46    
47    /**
48     * Reads a fetch.txt file
49     * 
50     * @param fetchFile the specific fetch file
51     * @param encoding the encoding to read the file with
52     * @param bagRootDir the root directory of the bag
53     * @return a list of items to fetch
54     * 
55     * @throws IOException if there is a problem reading a file
56     * @throws MaliciousPathException if the path was crafted to point outside the bag directory
57     * @throws InvalidBagitFileFormatException if the fetch format does not follow the bagit specification
58     */
59    @SuppressWarnings("PMD.AvoidInstantiatingObjectsInLoops")
60    public static List<FetchItem> readFetch(final Path fetchFile, final Charset encoding, final Path bagRootDir) throws IOException, MaliciousPathException, InvalidBagitFileFormatException{
61      logger.info(messages.getString("reading_fetch_file"), fetchFile);
62      final List<FetchItem> itemsToFetch = new ArrayList<>();
63      
64      try(final BufferedReader reader = Files.newBufferedReader(fetchFile, encoding)){
65        String line = reader.readLine();
66        String[] parts = null;
67        long length = 0;
68        URL url = null;
69        while(line != null){
70          if(line.matches(FETCH_LINE_REGEX) && !line.matches("\\s*")){
71            parts = line.split("\\s+", 3);
72            final Path path = TagFileReader.createFileFromManifest(bagRootDir, parts[2]);
73            length = parts[1].equals("-") ? -1 : Long.decode(parts[1]);
74            url = new URL(parts[0]);
75            
76            logger.debug(messages.getString("read_fetch_file_line"), url, length, parts[2], fetchFile);
77            final FetchItem itemToFetch = new FetchItem(url, length, path);
78            itemsToFetch.add(itemToFetch);
79          }
80          else{
81            throw new InvalidBagitFileFormatException(messages.getString("invalid_fetch_file_line_error").replace("{}", line));
82          }
83          
84          line = reader.readLine();
85        }
86      }
87  
88      return itemsToFetch;
89    }
90  }
