16  package nl.knaw.dans.bagit.verify;
17  
18  import java.io.IOException;
19  import java.nio.file.Files;
20  import java.nio.file.Path;
21  import java.util.ResourceBundle;
22  import java.util.AbstractMap.SimpleImmutableEntry;
23  
24  import nl.knaw.dans.bagit.domain.Bag;
25  import nl.knaw.dans.bagit.exceptions.InvalidPayloadOxumException;
26  import nl.knaw.dans.bagit.exceptions.PayloadOxumDoesNotExistException;
27  import org.slf4j.Logger;
28  import org.slf4j.LoggerFactory;
29  import org.slf4j.helpers.MessageFormatter;
30  
31  import nl.knaw.dans.bagit.util.PathUtils;
32  
33  /**
34   * responsible for all things related to quick verification. Quick verification does not
35   * mean that a Bag is valid, only that a cursory check has been made. For a full verification
36   * see {@link BagVerifier}
37   */
38  public final class QuickVerifier {
39    private static final Logger logger = LoggerFactory.getLogger(QuickVerifier.class);
40    private static final ResourceBundle messages = ResourceBundle.getBundle("MessageBundle");
41    private static final String PAYLOAD_OXUM_REGEX = "\\d+\\.\\d+";
42    
43    private QuickVerifier(){
44      //intentionally left empty
45    }
46    
47    /**
48     * Determine if we can quickly verify by comparing the number of files and the total number of bytes expected
49     * 
50     * @param bag the {@link Bag} object you wish to check
51     * @return true if the bag can be quickly verified
52     */
53    public static boolean canQuickVerify(final Bag bag){
54      final String payloadOxum = getPayloadOxum(bag);
55      logger.debug(messages.getString("found_payload_oxum"), payloadOxum, bag.getRootDir());
56      return payloadOxum != null && payloadOxum.matches(PAYLOAD_OXUM_REGEX) && bag.getItemsToFetch().size() == 0;
57    }
58    
59    /*
60     * Get the Payload-Oxum value from the key value pairs
61     */
62    private static String getPayloadOxum(final Bag bag){
63      for(final SimpleImmutableEntry<String,String> keyValue : bag.getMetadata().getAll()){
64        if("Payload-Oxum".equals(keyValue.getKey())){
65          return keyValue.getValue();
66        }
67      }
68      return null;
69    }
70    
71    /**
72     * Quickly verify by comparing the number of files and the total number of bytes expected
73     * 
74     * @param bag the bag to verify by payload-oxum
75     * 
76     * @throws IOException if there is an error reading a file
77     * @throws InvalidPayloadOxumException if either the total bytes or the number of files
78     * calculated for the payload directory of the bag is different than the supplied values
79     * @throws PayloadOxumDoesNotExistException if the bag does not contain a payload-oxum.
80     * To check, run {@link BagVerifier#canQuickVerify}
81     */
82    public static void quicklyVerify(final Bag bag) throws IOException, InvalidPayloadOxumException{
83      final String payloadOxum = getPayloadOxum(bag);
84      if(payloadOxum == null || !payloadOxum.matches(PAYLOAD_OXUM_REGEX)){
85        throw new PayloadOxumDoesNotExistException(messages.getString("payload_oxum_missing_error"));
86      }
87  
88      final String[] parts = payloadOxum.split("\\.");
89      logger.debug(messages.getString("parse_size_in_bytes"), parts[0]);
90      final long totalSize = Long.parseLong(parts[0]);
91      logger.debug(messages.getString("parse_number_of_files"), parts[1]);
92      final long numberOfFiles = Long.parseLong(parts[1]);
93      
94      final Path payloadDir = PathUtils.getDataDir(bag);
95      final FileCountAndTotalSizeVistor vistor = new FileCountAndTotalSizeVistor();
96      Files.walkFileTree(payloadDir, vistor);
97      logger.debug(messages.getString("compare_payload_oxums"), payloadOxum, vistor.getTotalSize(), vistor.getCount(), payloadDir);
98      
99      if(totalSize != vistor.getTotalSize()){
100       final String formattedMessage = messages.getString("invalid_total_size_error");
101       throw new InvalidPayloadOxumException(MessageFormatter.format(formattedMessage, totalSize, vistor.getTotalSize()).getMessage());
102     }
103     if(numberOfFiles != vistor.getCount()){
104       final String formattedMessage = messages.getString("invalid_file_cound_error");
105       throw new InvalidPayloadOxumException(MessageFormatter.format(formattedMessage, numberOfFiles, vistor.getCount()).getMessage());
106     }
107   }
108 }
