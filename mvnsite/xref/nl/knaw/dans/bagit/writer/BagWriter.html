18  import java.io.IOException;
19  import java.nio.file.Files;
20  import java.nio.file.Path;
21  import java.security.MessageDigest;
22  import java.security.NoSuchAlgorithmException;
23  import java.util.HashSet;
24  import java.util.Map.Entry;
25  import java.util.ResourceBundle;
26  import java.util.Set;
27  
28  import nl.knaw.dans.bagit.domain.Bag;
29  import nl.knaw.dans.bagit.domain.Manifest;
30  import nl.knaw.dans.bagit.hash.Hasher;
31  import nl.knaw.dans.bagit.util.PathUtils;
32  import org.slf4j.Logger;
33  import org.slf4j.LoggerFactory;

35  /**
36   * responsible for writing out a {@link Bag}
37   */
38  public final class BagWriter {
39    private static final Logger logger = LoggerFactory.getLogger(BagWriter.class);
40    private static final ResourceBundle messages = ResourceBundle.getBundle("MessageBundle");
41  
42    private BagWriter(){
43      //intentionally left empty
44    }
45    
46    /**
47     * Write the bag out to the specified directory. 
48     * If an error occurs some of the files may have been written out to the filesystem.
49     * tag manifest(s) are updated prior to writing to ensure bag is valid after completion, 
50     * it is therefore recommended if you are going to further interact with the bag to read it from specified outputDir path
51     * 
52     * @param bag the {@link Bag} object to write out
53     * @param outputDir the output directory that will become the root of the bag
54     * 
55     * @throws IOException if there is a problem writing a file
56     * @throws NoSuchAlgorithmException when trying to generate a {@link MessageDigest} which is used during update.
57     */
58    public static void write(final Bag bag, final Path outputDir) throws IOException, NoSuchAlgorithmException{
59      logger.debug(messages.getString("writing_payload_files"));
60      final Path bagitDir = PayloadWriter.writeVersionDependentPayloadFiles(bag, outputDir);
61      
62      logger.debug(messages.getString("upsert_payload_oxum"));
63      final String payloadOxum = PathUtils.generatePayloadOxum(PathUtils.getDataDir(bag.getVersion(), outputDir));
64      bag.getMetadata().upsertPayloadOxum(payloadOxum);
65      
66      logger.debug(messages.getString("writing_bagit_file"));
67      BagitFileWriter.writeBagitFile(bag.getVersion(), bag.getFileEncoding(), bagitDir);
68      
69      logger.debug(messages.getString("writing_payload_manifests"));
70      ManifestWriter.writePayloadManifests(bag.getPayLoadManifests(), bagitDir, bag.getRootDir(), bag.getFileEncoding());
71  
72      if(!bag.getMetadata().isEmpty()){
73        logger.debug(messages.getString("writing_bag_metadata"));
74        MetadataWriter.writeBagMetadata(bag.getMetadata(), bag.getVersion(), bagitDir, bag.getFileEncoding());
75      }
76      if(bag.getItemsToFetch().size() > 0){
77        logger.debug(messages.getString("writing_fetch_file"));
78        FetchWriter.writeFetchFile(bag.getItemsToFetch(), bagitDir, bag.getRootDir(), bag.getFileEncoding());
79      }
80      if(bag.getTagManifests().size() > 0){
81        logger.debug(messages.getString("writing_tag_manifests"));
82        writeTagManifestFiles(bag.getTagManifests(), bagitDir, bag.getRootDir());
83        final Set<Manifest> updatedTagManifests = updateTagManifests(bag, outputDir);
84        bag.setTagManifests(updatedTagManifests);
85        ManifestWriter.writeTagManifests(updatedTagManifests, bagitDir, outputDir, bag.getFileEncoding());
86      }
87    }
88    
89    /*
90     * Update the tag manifest cause the checksum of the other tag files will have changed since we just wrote them out to disk
91     */
92    @SuppressWarnings("PMD.AvoidInstantiatingObjectsInLoops")
93    private static Set<Manifest> updateTagManifests(final Bag bag, final Path newBagRootDir) throws NoSuchAlgorithmException, IOException{
94      final Set<Manifest> newManifests = new HashSet<>();
95      
96      for(final Manifest tagManifest : bag.getTagManifests()){
97        final Manifest newManifest = new Manifest(tagManifest.getAlgorithm());
98        
99        for(final Path originalPath : tagManifest.getFileToChecksumMap().keySet()){
100         final Path relativePath = bag.getRootDir().relativize(originalPath);
101         final Path pathToUpdate = newBagRootDir.resolve(relativePath);
102         final MessageDigest messageDigest = MessageDigest.getInstance(tagManifest.getAlgorithm().getMessageDigestName());
103         final String newChecksum = Hasher.hash(pathToUpdate, messageDigest);
104         newManifest.getFileToChecksumMap().put(pathToUpdate, newChecksum);
105       }
106       
107       newManifests.add(newManifest);
108     }
109     
110     return newManifests;
111   }
112   
113   /*
114    * Write the tag manifest files
115    */
116   private static void writeTagManifestFiles(final Set<Manifest> manifests, final Path outputDir, final Path bagRootDir) throws IOException{
117     for(final Manifest manifest : manifests){
118       for(final Entry<Path, String> entry : manifest.getFileToChecksumMap().entrySet()){
119         final Path relativeLocation = bagRootDir.relativize(entry.getKey());
120         final Path writeTo = outputDir.resolve(relativeLocation);
121         final Path writeToParent = writeTo.getParent();
122         if(!Files.exists(writeTo) && writeToParent != null){
123           Files.createDirectories(writeToParent);
124           Files.copy(entry.getKey(), writeTo);
125         }
126       }
127     }
128   }
