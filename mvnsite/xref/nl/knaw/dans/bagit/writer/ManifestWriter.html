16  package nl.knaw.dans.bagit.writer;
17  
18  import java.io.IOException;
19  import java.nio.charset.Charset;
20  import java.nio.file.Files;
21  import java.nio.file.Path;
22  import java.nio.file.StandardOpenOption;
23  import java.util.Map.Entry;
24  import java.util.ResourceBundle;
25  import java.util.Set;
26  
27  import nl.knaw.dans.bagit.domain.Manifest;
28  import org.slf4j.Logger;
29  import org.slf4j.LoggerFactory;
30  
31  import java.io.BufferedWriter;
32  
33  /**
34   * Responsible for writing out a {@link Manifest} to the filesystem
35   */
36  public final class ManifestWriter{
37    private static final Logger logger = LoggerFactory.getLogger(PayloadWriter.class);
38    private static final ResourceBundle messages = ResourceBundle.getBundle("MessageBundle");
39    
40    private ManifestWriter(){
41      //intentionally left empty
42    }
43    
44    /**
45     * Write the payload <b>manifest(s)</b> to the output directory
46     * 
47     * @param manifests the payload{@link Manifest}s to write out
48     * @param outputDir the root of where the manifest is being written to
49     * @param bagitRootDir the path to the root of the bag
50     * @param charsetName the name of the encoding for the file
51     * 
52     * @throws IOException if there was a problem writing a file
53     */
54    public static void writePayloadManifests(final Set<Manifest> manifests, final Path outputDir, final Path bagitRootDir, final Charset charsetName) throws IOException{
55      logger.info(messages.getString("writing_payload_manifests"));
56      writeManifests(manifests, outputDir, bagitRootDir, "manifest-", charsetName);
57    }
58    
59    /**
60     * Write the tag <b>manifest(s)</b> to the output directory
61     * 
62     * @param tagManifests the tag{@link Manifest}s to write out
63     * @param outputDir the root of where the manifest is being written to
64     * @param bagitRootDir the path to the root of the bag
65     * @param charsetName the name of the encoding for the file
66     * 
67     * @throws IOException if there was a problem writing a file
68     */
69    public static void writeTagManifests(final Set<Manifest> tagManifests, final Path outputDir, final Path bagitRootDir, final Charset charsetName) throws IOException{
70      logger.info(messages.getString("writing_tag_manifests"));
71      writeManifests(tagManifests, outputDir, bagitRootDir, "tagmanifest-", charsetName);
72    }
73    
74    /*
75     * Generic method to write manifests
76     */
77    private static void writeManifests(final Set<Manifest> manifests, final Path outputDir, final Path relativeTo, final String filenameBase, final Charset charsetName) throws IOException{
78      for(final Manifest manifest : manifests){
79        final Path manifestPath = outputDir.resolve(filenameBase + manifest.getAlgorithm().getBagitName() + ".txt");
80        logger.debug(messages.getString("writing_manifest_to_path"), manifestPath);
81  
82        Files.deleteIfExists(manifestPath);
83        Files.createFile(manifestPath);
84  
85        try (BufferedWriter writer = Files.newBufferedWriter(manifestPath, charsetName,
86              StandardOpenOption.APPEND, StandardOpenOption.CREATE)) {
87          for (final Entry<Path, String> entry : manifest.getFileToChecksumMap().entrySet()) {
88            //there are 2 spaces between the checksum and the path so that the manifests are compatible with the md5sum tools available on most unix systems.
89            //This may cause problems on windows due to it being text mode, in which case either replace with a * or try verifying in binary mode with --binary
90            final String line = entry.getValue() + "  " + RelativePathWriter.formatRelativePathString(relativeTo, entry.getKey());
91            logger.debug(messages.getString("writing_line_to_file"), line, manifestPath);
92            writer.append(line);
93          }
94        }
95      }
96    }
97  }
