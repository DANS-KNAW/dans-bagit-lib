16  package nl.knaw.dans.bagit.writer;
17  
18  import java.io.IOException;
19  import java.nio.charset.Charset;
20  import java.nio.file.Files;
21  import java.nio.file.Path;
22  import java.nio.file.StandardOpenOption;
23  import java.util.ResourceBundle;
24  import java.util.AbstractMap.SimpleImmutableEntry;
25  
26  import nl.knaw.dans.bagit.domain.Metadata;
27  import nl.knaw.dans.bagit.domain.Version;
28  import org.slf4j.Logger;
29  import org.slf4j.LoggerFactory;
30  
31  /**
32   * Responsible for writing out the bag {@link Metadata} to the filesystem
33   */
34  public final class MetadataWriter {
35    private static final Logger logger = LoggerFactory.getLogger(MetadataWriter.class);
36    private static final Version VERSION_0_95 = new Version(0, 95);
37    private static final ResourceBundle messages = ResourceBundle.getBundle("MessageBundle");
38  
39    private MetadataWriter(){
40      //intentionall left empty
41    }
42    
43    /**
44     * Write the bag-info.txt (or package-info.txt) file to the specified outputDir with specified encoding (charsetName)
45     * 
46     * @param metadata the key value pair info in the bag-info.txt file
47     * @param version the version of the bag you are writing
48     * @param outputDir the root of the bag
49     * @param charsetName the name of the encoding for the file
50     * 
51     * @throws IOException if there was a problem writing a file
52     */
53    public static void writeBagMetadata(final Metadata metadata, final Version version, final Path outputDir, final Charset charsetName) throws IOException{
54      Path bagInfoFilePath = outputDir.resolve("bag-info.txt");
55      if(version.isSameOrOlder(VERSION_0_95)){
56        bagInfoFilePath = outputDir.resolve("package-info.txt");
57      }
58      logger.debug(messages.getString("writing_metadata_to_path"), bagInfoFilePath.getFileName(), outputDir);
59  
60      Files.deleteIfExists(bagInfoFilePath);
61      final StringBuilder lines = new StringBuilder();
62      
63      for(final SimpleImmutableEntry<String, String> entry : metadata.getAll()){
64        final String key = entry.getKey();
65        String value = entry.getValue();
66        value = value.replaceAll("[^\\x09\\x20-\\x7E\\x0A\\x0D\\x80-\\uFFFF]", "");
67        lines.append(formatLine(key, value, version)).append(System.lineSeparator());
68      }
69      
70      logger.debug(messages.getString("writing_line_to_file"), lines.toString(), bagInfoFilePath);
71      Files.write(bagInfoFilePath, lines.toString().getBytes(charsetName), 
72          StandardOpenOption.APPEND, StandardOpenOption.CREATE);
73    }
74  
75    private static String formatLine(final String key, final String value, final Version version) {
76      if (version.isSameOrOlder(VERSION_0_95)) {
77        final String fullLine = key + ": " + value;
78        return fullLine.replaceAll("(\\r\\n|\\r|\\n)", "$1 ");
79      }
80      
81      final StringBuilder sb = new StringBuilder();
82      final int keyPrefixLength = key.length() + 2; // "Key: "
83      sb.append(key).append(": ");
84      final String[] parts = value.split("(\\r\\n|\\r|\\n)", -1);
85      
86      for (int i = 0; i < parts.length; i++) {
87        String line = parts[i];
88        if (i > 0) {
89          sb.append(System.lineSeparator());
90          line = " " + line;
91        }
92        
93        final int maxLength = i == 0 ? 79 - keyPrefixLength : 78; // Continuation lines have " " prefix
94        // Check if it's already "well-wrapped" or short
95        if (line.length() > maxLength) {
96          sb.append(wrapLine(line, i == 0, keyPrefixLength));
97        } else {
98          sb.append(line);
99        }
100     }
101     return sb.toString();
102   }
103 
104   private static String wrapLine(final String line, final boolean isFirstLine, final int keyPrefixLength) {
105     final StringBuilder sb = new StringBuilder();
106     int start = 0;
107     boolean firstWrap = true;
108     
109     while (start < line.length()) {
110       int maxLength = 78; // Indented line starts with " ", so 78 more chars = 79
111       if (firstWrap && isFirstLine) {
112         maxLength = 79 - keyPrefixLength;
113       }
114       
115       int end = Math.min(start + maxLength, line.length());
116       
117       if (end < line.length()) {
118         int lastSpace = line.lastIndexOf(' ', end);
119         if (lastSpace > start) {
120           end = lastSpace;
121         }
122       }
123       
124       String sub = line.substring(start, end);
125       if (start > 0) {
126         sb.append(System.lineSeparator());
127         sb.append(" ");
128       }
129       sb.append(sub);
130       
131       start = end;
132       while (start < line.length() && line.charAt(start) == ' ') {
133         start++;
134       }
135       firstWrap = false;
136     }
137     
138     return sb.toString();
139   }
