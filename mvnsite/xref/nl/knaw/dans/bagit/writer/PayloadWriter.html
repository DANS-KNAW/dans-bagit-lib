16  package nl.knaw.dans.bagit.writer;
17  
18  import java.io.IOException;
19  import java.nio.file.Files;
20  import java.nio.file.Path;
21  import java.nio.file.StandardCopyOption;
22  import java.util.HashSet;
23  import java.util.List;
24  import java.util.ResourceBundle;
25  import java.util.Set;
26  
27  import nl.knaw.dans.bagit.domain.Bag;
28  import nl.knaw.dans.bagit.domain.FetchItem;
29  import nl.knaw.dans.bagit.domain.Manifest;
30  import nl.knaw.dans.bagit.domain.Version;
31  import org.slf4j.Logger;
32  import org.slf4j.LoggerFactory;
33  
34  /**
35   * Responsible for writing out the bag payload to the filesystem
36   */
37  public final class PayloadWriter {
38    private static final Logger logger = LoggerFactory.getLogger(PayloadWriter.class);
39    private static final Version VERSION_2_0 = new Version(2, 0);
40    private static final ResourceBundle messages = ResourceBundle.getBundle("MessageBundle");
41    
42    private PayloadWriter(){
43      //intentionally left empty
44    }
45    
46    /*
47     * Write the payload files in the data directory or under the root directory depending on the version
48     */
49    static Path writeVersionDependentPayloadFiles(final Bag bag, final Path outputDir) throws IOException{
50      Path bagitDir = outputDir;
51      //@Incubating
52      if(bag.getVersion().isSameOrNewer(VERSION_2_0)){
53        bagitDir = outputDir.resolve(".bagit");
54        Files.createDirectories(bagitDir);
55        writePayloadFiles(bag.getPayLoadManifests(), bag.getItemsToFetch(), outputDir, bag.getRootDir());
56      }
57      else{
58        final Path dataDir = outputDir.resolve("data");
59        Files.createDirectories(dataDir);
60        writePayloadFiles(bag.getPayLoadManifests(), bag.getItemsToFetch(), dataDir, bag.getRootDir().resolve("data"));
61      }
62      
63      return bagitDir;
64    }
65    
66    /**
67    * Write the payload <b>file(s)</b> to the output directory
68    * 
69    * @param payloadManifests the set of objects representing the payload manifests
70    * @param fetchItems the list of items to exclude from writing in the output directory because they will be fetched
71    * @param outputDir the data directory of the bag
72    * @param bagDataDir the data directory of the bag
73    * 
74    * @throws IOException if there was a problem writing a file
75    */
76   public static void writePayloadFiles(final Set<Manifest> payloadManifests, final List<FetchItem> fetchItems, final Path outputDir, final Path bagDataDir) throws IOException{
77     logger.info(messages.getString("writing_payload_files"));
78     final Set<Path> fetchPaths = getFetchPaths(fetchItems, bagDataDir);
79     
80     for(final Manifest payloadManifest : payloadManifests){
81       for(final Path payloadFile : payloadManifest.getFileToChecksumMap().keySet()){
82         final Path relativePayloadPath = bagDataDir.relativize(payloadFile);
83         
84         if(fetchPaths.contains(relativePayloadPath.normalize())) {
85           logger.info(messages.getString("skip_fetch_item_when_writing_payload"), payloadFile);
86         }
87         else {
88           final Path writeToPath = outputDir.resolve(relativePayloadPath);
89           logger.debug(messages.getString("writing_payload_file_to_path"), payloadFile, writeToPath);
90           final Path parent = writeToPath.getParent();
91           if(parent != null){
92             Files.createDirectories(parent);
93           }
94           Files.copy(payloadFile, writeToPath, StandardCopyOption.COPY_ATTRIBUTES, StandardCopyOption.REPLACE_EXISTING);
95         }
96       }
97     }
98   }
99   
100   private static Set<Path> getFetchPaths(final List<FetchItem> fetchItems, final Path bagDataDir) {
101     final Set<Path> fetchPaths = new HashSet<>();
102     for (final FetchItem fetchItem : fetchItems) {
103       final Path parent = bagDataDir.getParent();
104       if(parent != null){
105         fetchPaths.add(bagDataDir.relativize(parent.resolve(fetchItem.getPath())));
106       }
107     }
108     return fetchPaths;
109  }
110 }
