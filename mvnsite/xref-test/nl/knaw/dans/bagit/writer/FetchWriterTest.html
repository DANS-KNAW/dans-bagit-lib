16  package nl.knaw.dans.bagit.writer;
17  
18  import java.lang.reflect.InvocationTargetException;
19  import java.net.URL;
20  import java.nio.charset.StandardCharsets;
21  import java.nio.file.Files;
22  import java.nio.file.Path;
23  import java.util.ArrayList;
24  import java.util.Arrays;
25  import java.util.List;
26  
27  import nl.knaw.dans.bagit.PrivateConstructorTest;
28  import nl.knaw.dans.bagit.domain.FetchItem;
29  import org.junit.jupiter.api.Assertions;
30  import org.junit.jupiter.api.Test;
31  
32  public class FetchWriterTest extends PrivateConstructorTest {
33    
34    @Test
35    public void testClassIsWellDefined() throws NoSuchMethodException, InvocationTargetException, InstantiationException, IllegalAccessException{
36      assertUtilityClassWellDefined(FetchWriter.class);
37    }
38  
39    @Test
40    public void testWriteFetchFile() throws Exception{
41      Path rootPath = createDirectory("fetchTest");
42      Path fetch = rootPath.resolve("fetch.txt");
43      URL url = new URL("http://localhost:/foo/bar");
44      List<FetchItem> itemsToFetch = Arrays.asList(new FetchItem(url, -1l, rootPath.resolve("/data/foo/bar")),
45          new FetchItem(url, 100l, rootPath.resolve("/data/foo/bar")));
46      
47      
48      Assertions.assertFalse(Files.exists(fetch));
49      FetchWriter.writeFetchFile(itemsToFetch, rootPath, rootPath, StandardCharsets.UTF_8);
50      Assertions.assertTrue(Files.exists(fetch));
51    }
52    
53    @Test
54    public void testFetchFileIsFormattedCorrectly() throws Exception{
55      Path rootPath = createDirectory("fetchFormatted");
56      Path fetch = rootPath.resolve("fetch.txt");
57      List<FetchItem> itemsToFetch = new ArrayList<>();
58      
59      itemsToFetch.add(new FetchItem(new URL("http://localhost:8989/bags/v0_96/holey-bag/data/dir1/test3.txt"), null, rootPath.resolve("data/dir1/test3.txt")));
60      itemsToFetch.add(new FetchItem(new URL("http://localhost:8989/bags/v0_96/holey-bag/data/dir2/dir3/test5.txt"), null, rootPath.resolve("data/dir2/dir3/test5.txt")));
61      itemsToFetch.add(new FetchItem(new URL("http://localhost:8989/bags/v0_96/holey-bag/data/dir2/test4.txt"), null, rootPath.resolve("data/dir2/test4.txt")));
62      itemsToFetch.add(new FetchItem(new URL("http://localhost:8989/bags/v0_96/holey-bag/data/test%201.txt"), null, rootPath.resolve("data/test 1.txt")));
63      itemsToFetch.add(new FetchItem(new URL("http://localhost:8989/bags/v0_96/holey-bag/data/test2.txt"), null, rootPath.resolve("data/test2.txt")));
64      
65      FetchWriter.writeFetchFile(itemsToFetch, rootPath, rootPath, StandardCharsets.UTF_8);
66      
67      List<String> expectedLines = Arrays.asList("http://localhost:8989/bags/v0_96/holey-bag/data/dir1/test3.txt - data/dir1/test3.txt", 
68          "http://localhost:8989/bags/v0_96/holey-bag/data/dir2/dir3/test5.txt - data/dir2/dir3/test5.txt", 
69          "http://localhost:8989/bags/v0_96/holey-bag/data/dir2/test4.txt - data/dir2/test4.txt", 
70          "http://localhost:8989/bags/v0_96/holey-bag/data/test%201.txt - data/test 1.txt",
71          "http://localhost:8989/bags/v0_96/holey-bag/data/test2.txt - data/test2.txt");
72      List<String> actualLines = Files.readAllLines(fetch);
73      
74      Assertions.assertEquals(expectedLines, actualLines);
75    }
76  }
