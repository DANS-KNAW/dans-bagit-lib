63    @Test
64    public void testWriteAndReadMultilineMetadata() throws Exception {
65      Path rootDir = createDirectory("multilineTest");
66      Metadata metadata = new Metadata();
67      String multilineValue = "This is a" + System.lineSeparator() + "multi-line" + System.lineSeparator() + "value.";
68      metadata.add("Description", multilineValue);
69      metadata.add("Contact-Name", "John Doe");
70  
71      MetadataWriter.writeBagMetadata(metadata, new Version(1, 0), rootDir, StandardCharsets.UTF_8);
72  
73      Path bagInfo = rootDir.resolve("bag-info.txt");
74      String content = Files.readString(bagInfo, StandardCharsets.UTF_8);
75  
76      // Check if subsequent lines are indented
77      String[] lines = content.split("\\R");
78      assertTrue(lines[0].startsWith("Description: "));
79      assertTrue(lines[1].startsWith(" "), "Line 2 should be indented");
80      assertTrue(lines[2].startsWith(" "), "Line 3 should be indented");
81  
82      // Read it back
83      List<SimpleImmutableEntry<String, String>> readMetadata = KeyValueReader.readKeyValuesFromFile(bagInfo, ":", StandardCharsets.UTF_8);
84      
85      boolean foundDescription = false;
86      for (SimpleImmutableEntry<String, String> entry : readMetadata) {
87        if ("Description".equals(entry.getKey())) {
88          Assertions.assertEquals(multilineValue, entry.getValue());
89          foundDescription = true;
90        }
91      }
92      assertTrue(foundDescription);
93    }

95    @Test
96    public void testSanitizeMetadata() throws Exception {
97      Path rootDir = createDirectory("sanitizeTest");
98      Metadata metadata = new Metadata();
99      // \u0000 is a non-printable char, \u0007 is bell
100     String dirtyValue = "Value with\u0000 non-printable\u0007 chars.";
101     String cleanValue = "Value with non-printable chars.";
102     metadata.add("Custom-Key", dirtyValue);
103 
104     MetadataWriter.writeBagMetadata(metadata, new Version(1, 0), rootDir, StandardCharsets.UTF_8);
105 
106     Path bagInfo = rootDir.resolve("bag-info.txt");
107     List<SimpleImmutableEntry<String, String>> readMetadata = KeyValueReader.readKeyValuesFromFile(bagInfo, ":", StandardCharsets.UTF_8);
108 
109     boolean found = false;
110     for (SimpleImmutableEntry<String, String> entry : readMetadata) {
111       if ("Custom-Key".equals(entry.getKey())) {
112         Assertions.assertEquals(cleanValue, entry.getValue());
113         found = true;
114       }
115     }
116     assertTrue(found);

119   @Test
120   public void testWrapLongLines() throws Exception {
121     Path rootDir = createDirectory("wrapLongLinesTest");
122     Metadata metadata = new Metadata();
123     String longValue = "This is a very long value that should definitely exceed the seventy-nine characters limit that is recommended by the BagIt RFC 8493 section two point two point two.";
124     // Length is ~166 chars.
125     metadata.add("Long-Key", longValue);
126 
127     MetadataWriter.writeBagMetadata(metadata, new Version(1, 0), rootDir, StandardCharsets.UTF_8);
128 
129     Path bagInfo = rootDir.resolve("bag-info.txt");
130     String content = Files.readString(bagInfo, StandardCharsets.UTF_8);
131 
132     String[] lines = content.split("\\R");
133     for (String line : lines) {
134       assertTrue(line.length() <= 79, "Line length should be <= 79: " + line.length());
135       if (!line.startsWith("Long-Key: ")) {
136         assertTrue(line.startsWith(" "), "Wrapped lines should be indented");
137       }
138     }
139 
140     // Read it back
141     List<SimpleImmutableEntry<String, String>> readMetadata = KeyValueReader.readKeyValuesFromFile(bagInfo, ":", StandardCharsets.UTF_8);
142     boolean foundLongKey = false;
143     for (SimpleImmutableEntry<String, String> entry : readMetadata) {
144       if ("Long-Key".equals(entry.getKey())) {
145         // Since it's wrapped at spaces, the space is replaced by newline in reading if it's joined by newline
146         // But wrapLine also preserves spaces? Let's check what it actually produces.
147         // It should match the longValue with some spaces replaced by newlines.
148         Assertions.assertEquals(longValue, entry.getValue().replace(System.lineSeparator(), " "));
149         foundLongKey = true;
150       }
151     }
152     assertTrue(foundLongKey);
153   }

155   @Test
156   public void testMultilineAndWrapping() throws Exception {
157     Path rootDir = createDirectory("multilineAndWrappingTest");
158     Metadata metadata = new Metadata();
159     // A multiline value where the first line is short and the second is long
160     String value = "Short first line\nThis is a very long second line that will definitely need wrapping because it's much longer than seventy-nine characters.";
161     metadata.add("Description", value);
162 
163     MetadataWriter.writeBagMetadata(metadata, new Version(1, 0), rootDir, StandardCharsets.UTF_8);
164 
165     Path bagInfo = rootDir.resolve("bag-info.txt");
166     String content = Files.readString(bagInfo, StandardCharsets.UTF_8);
167 
168     // Check for double newlines or lines containing only a space
169     String[] lines = content.split("\\r?\\n");
170     for (int i = 0; i < lines.length; i++) {
171       String line = lines[i];
172       Assertions.assertFalse(line.isEmpty() && i < lines.length - 1, "Should not have empty lines in the middle (caused by double newlines)");
173       if (i > 0) {
174         // All lines after the first one of an entry must be indented
175         assertTrue(line.startsWith(" "), "Continuation line must start with a space: [" + line + "]");
176         // If it was a double newline, we'd see a line that is just " " followed by another line
177         Assertions.assertNotEquals(" ", line, "Should not have a line that is just a single space");
178       }
179     }
