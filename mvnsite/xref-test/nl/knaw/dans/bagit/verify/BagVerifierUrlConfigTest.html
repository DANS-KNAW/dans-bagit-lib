63      @Test
64      public void testUrlSpecificHeaders() throws Exception {
65          final String content = "test content";
66          final String globalHeaderName = "X-Global";
67          final String globalHeaderValue = "global-val";
68          final String urlHeaderName = "X-Url-Specific";
69          final String urlHeaderValue = "url-val";
70          
71          final AtomicBoolean globalReceived = new AtomicBoolean(false);
72          final AtomicBoolean urlSpecificReceived = new AtomicBoolean(false);
73  
74          server.createContext("/specific", new HttpHandler() {
75              @Override
76              public void handle(HttpExchange exchange) throws IOException {
77                  if (globalHeaderValue.equals(exchange.getRequestHeaders().getFirst(globalHeaderName))) {
78                      globalReceived.set(true);
79                  }
80                  if (urlHeaderValue.equals(exchange.getRequestHeaders().getFirst(urlHeaderName))) {
81                      urlSpecificReceived.set(true);
82                  }
83                  
84                  byte[] response = content.getBytes(StandardCharsets.UTF_8);
85                  exchange.sendResponseHeaders(200, response.length);
86                  try (OutputStream os = exchange.getResponseBody()) {
87                      os.write(response);
88                  }
89              }
90          });
91  
92          Path bagDir = Files.createTempDirectory(folder, "url-config-bag");
93          Files.write(bagDir.resolve("bagit.txt"), "BagIt-Version: 0.97\nTag-File-Character-Encoding: UTF-8\n".getBytes());
94          Files.createDirectory(bagDir.resolve("data"));
95          
96          MessageDigest sha1 = MessageDigest.getInstance("SHA-1");
97          sha1.update(content.getBytes(StandardCharsets.UTF_8));
98          String hash = formatMessageDigest(sha1.digest());
99          
100         Files.write(bagDir.resolve("manifest-sha1.txt"), (hash + "  data/test.txt\n").getBytes());
101         
102         URL remoteUrl = new URL("http://localhost:" + port + "/specific");
103         Files.write(bagDir.resolve("fetch.txt"), (remoteUrl.toString() + " " + content.length() + " data/test.txt\n").getBytes());
104         
105         Bag bag = reader.read(bagDir);
106         
107         Map<String, String> globalHeaders = new HashMap<>();
108         globalHeaders.put(globalHeaderName, globalHeaderValue);
109         
110         Map<String, Map<String, String>> urlConfigs = new HashMap<>();
111         Map<String, String> specificHeaders = new HashMap<>();
112         specificHeaders.put(urlHeaderName, urlHeaderValue);
113         urlConfigs.put("http://localhost:" + port + "/specific", specificHeaders);
114         
115         sut.isValid(bag, true, true, globalHeaders, urlConfigs);
116 
117         Assertions.assertTrue(globalReceived.get(), "Global header should have been received");
118         Assertions.assertTrue(urlSpecificReceived.get(), "URL specific header should have been received");
119     }

121     @Test
122     public void testUrlPrefixMatching() throws Exception {
123         final String content = "test content";
124         final String headerName = "X-Url-Specific";
125         final String headerValue = "url-val";
126         
127         final AtomicBoolean headerReceived = new AtomicBoolean(false);
128 
129         server.createContext("/prefix/specific", new HttpHandler() {
130             @Override
131             public void handle(HttpExchange exchange) throws IOException {
132                 if (headerValue.equals(exchange.getRequestHeaders().getFirst(headerName))) {
133                     headerReceived.set(true);
134                 }
135                 
136                 byte[] response = content.getBytes(StandardCharsets.UTF_8);
137                 exchange.sendResponseHeaders(200, response.length);
138                 try (OutputStream os = exchange.getResponseBody()) {
139                     os.write(response);
140                 }
141             }
142         });
143 
144         Path bagDir = Files.createTempDirectory(folder, "url-prefix-bag");
145         Files.write(bagDir.resolve("bagit.txt"), "BagIt-Version: 0.97\nTag-File-Character-Encoding: UTF-8\n".getBytes());
146         Files.createDirectory(bagDir.resolve("data"));
147         
148         MessageDigest sha1 = MessageDigest.getInstance("SHA-1");
149         sha1.update(content.getBytes(StandardCharsets.UTF_8));
150         String hash = formatMessageDigest(sha1.digest());
151         
152         Files.write(bagDir.resolve("manifest-sha1.txt"), (hash + "  data/test.txt\n").getBytes());
153         
154         URL remoteUrl = new URL("http://localhost:" + port + "/prefix/specific");
155         Files.write(bagDir.resolve("fetch.txt"), (remoteUrl.toString() + " " + content.length() + " data/test.txt\n").getBytes());
156         
157         Bag bag = reader.read(bagDir);
158         
159         Map<String, Map<String, String>> urlConfigs = new HashMap<>();
160         Map<String, String> specificHeaders = new HashMap<>();
161         specificHeaders.put(headerName, headerValue);
162         // Using a prefix
163         urlConfigs.put("http://localhost:" + port + "/prefix", specificHeaders);
164         
165         sut.isValid(bag, true, true, null, urlConfigs);
166 
167         Assertions.assertTrue(headerReceived.get(), "URL specific header should have been received because of prefix match");
168     }

170     @Test
171     public void testMultipleUrlConfigs() throws Exception {
172         final String content1 = "content 1";
173         final String content2 = "content 2";
174         final String header1 = "X-H1";
175         final String header2 = "X-H2";
176         
177         final AtomicBoolean h1Received = new AtomicBoolean(false);
178         final AtomicBoolean h2Received = new AtomicBoolean(false);
179 
180         server.createContext("/url1", new HttpHandler() {
181             @Override
182             public void handle(HttpExchange exchange) throws IOException {
183                 if ("v1".equals(exchange.getRequestHeaders().getFirst(header1))) h1Received.set(true);
184                 byte[] response = content1.getBytes(StandardCharsets.UTF_8);
185                 exchange.sendResponseHeaders(200, response.length);
186                 try (OutputStream os = exchange.getResponseBody()) { os.write(response); }
187             }
188         });
189         server.createContext("/url2", new HttpHandler() {
190             @Override
191             public void handle(HttpExchange exchange) throws IOException {
192                 if ("v2".equals(exchange.getRequestHeaders().getFirst(header2))) h2Received.set(true);
193                 byte[] response = content2.getBytes(StandardCharsets.UTF_8);
194                 exchange.sendResponseHeaders(200, response.length);
195                 try (OutputStream os = exchange.getResponseBody()) { os.write(response); }
196             }
197         });
198 
199         Path bagDir = Files.createTempDirectory(folder, "multi-url-bag");
200         Files.write(bagDir.resolve("bagit.txt"), "BagIt-Version: 0.97\nTag-File-Character-Encoding: UTF-8\n".getBytes());
201         Files.createDirectory(bagDir.resolve("data"));
202         
203         MessageDigest sha1 = MessageDigest.getInstance("SHA-1");
204         
205         sha1.update(content1.getBytes(StandardCharsets.UTF_8));
206         String hash1 = formatMessageDigest(sha1.digest());
207         sha1.reset();
208         sha1.update(content2.getBytes(StandardCharsets.UTF_8));
209         String hash2 = formatMessageDigest(sha1.digest());
210         
211         Files.write(bagDir.resolve("manifest-sha1.txt"), (hash1 + "  data/t1.txt\n" + hash2 + "  data/t2.txt\n").getBytes());
212         
213         URL url1 = new URL("http://localhost:" + port + "/url1");
214         URL url2 = new URL("http://localhost:" + port + "/url2");
215         Files.write(bagDir.resolve("fetch.txt"), (url1 + " " + content1.length() + " data/t1.txt\n" + url2 + " " + content2.length() + " data/t2.txt\n").getBytes());
216         
217         Bag bag = reader.read(bagDir);
218         
219         Map<String, Map<String, String>> urlConfigs = new HashMap<>();
220         Map<String, String> s1 = new HashMap<>(); s1.put(header1, "v1");
221         Map<String, String> s2 = new HashMap<>(); s2.put(header2, "v2");
222         urlConfigs.put(url1.toString(), s1);
223         urlConfigs.put(url2.toString(), s2);
224         
225         sut.isValid(bag, true, true, null, urlConfigs);
226 
227         Assertions.assertTrue(h1Received.get(), "Header 1 should have been received");
228         Assertions.assertTrue(h2Received.get(), "Header 2 should have been received");
229     }
230 
231     private String formatMessageDigest(final byte[] digest) {
232         StringBuilder sb = new StringBuilder();
233         for (byte b : digest) {
234             sb.append(String.format("%02x", b));
235         }
236         return sb.toString();
237     }
